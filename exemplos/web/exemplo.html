<script>
const ponte = new PonteImpressao({ colunas: 32 });

// 1. a ponte está no ar?
if (await ponte.online()) {

  // 2. o que esta máquina tem
  const lista = await ponte.impressoras();

  // 3. montar o cupom
  const L = PonteImpressao;
  const cupom = [
    L.centro('MINHA LOJA', 32),
    L.risco(32),
    L.linha('Coca-Cola 2L', '25,00', 32),
    L.linha('TOTAL', 'R$ 84,90', 32),
  ].join('\n');

  // 4. imprimir
  const r = await ponte.imprimir(cupom, { cortar: true });
  if (!r.ok) alert(r.erro);
}
</script>
